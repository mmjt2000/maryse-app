let currentSongIndex = 0;

function render${nombre}Playlist() {
    const container = document.getElementById("${nombre.toLowerCase()}Playlist");
    if (!container) return;
    
    container.innerHTML = ${nombre.toLowerCase()}FavSongs.map((song, idx) => \`
        <div class="song-item" onclick="playSong(\${idx})">
            <div class="song-info">
                <div class="song-title">🎵 \${song.title}</div>
                <div class="song-artist">\${song.artist}</div>
            </div>
            <div class="song-lang">\${song.language === 'en' ? '🇬🇧' : song.language === 'fr' ? '🇫🇷' : '🇪🇸'}</div>
            <div class="play-icon" onclick="event.stopPropagation(); window.open('\${song.url}', '_blank')">🎬</div>
        </div>
    \`).join('');
}

function playSong(index) {
    const song = ${nombre.toLowerCase()}FavSongs[index];
    if (!song) return;
    currentSongIndex = index;
    document.getElementById("nowPlayingTitle").innerHTML = \`🎵 \${song.title}\`;
    document.getElementById("nowPlayingArtist").innerHTML = \`\${song.artist} <span style="font-size:0.7rem;">(\${song.language === 'en' ? '🇬🇧 Inglés' : song.language === 'fr' ? '🇫🇷 Francés' : '🇪🇸 Español'})</span>\`;
    document.getElementById("nowPlayingLang").innerHTML = song.language === 'en' ? '🇬🇧' : song.language === 'fr' ? '🇫🇷' : '🇪🇸';
    document.querySelectorAll('.song-item').forEach((item, i) => { if (i === index) item.classList.add('active'); else item.classList.remove('active'); });
    if (song.url) window.open(song.url, '_blank');
}

function prevSong() { let newIndex = currentSongIndex - 1; if (newIndex < 0) newIndex = ${nombre.toLowerCase()}FavSongs.length - 1; playSong(newIndex); }
function nextSong() { let newIndex = currentSongIndex + 1; if (newIndex >= ${nombre.toLowerCase()}FavSongs.length) newIndex = 0; playSong(newIndex); }
function togglePlayPause() { if (${nombre.toLowerCase()}FavSongs[currentSongIndex]) window.open(${nombre.toLowerCase()}FavSongs[currentSongIndex].url, '_blank'); }
